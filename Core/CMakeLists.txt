cmake_minimum_required(VERSION 3.16)
project(BlondaCore CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(blonda_core STATIC src/bris.cpp src/drums.cpp src/wg.cpp src/engines.cpp src/seq3.cpp)
target_include_directories(blonda_core PUBLIC include)
target_compile_options(blonda_core PRIVATE -Wall -Wextra -O3)
add_executable(render tools/render.cpp)
target_link_libraries(render blonda_core)
target_compile_options(render PRIVATE -O3)
add_executable(probe tools/probe.cpp)
target_link_libraries(probe blonda_core)
add_executable(drumprobe tools/drumprobe.cpp)
target_link_libraries(drumprobe blonda_core)
add_executable(engprobe tools/engprobe.cpp)
target_link_libraries(engprobe blonda_core)
add_executable(comptest tools/comptest.cpp)
target_link_libraries(comptest blonda_core)
add_executable(convtest tools/convtest.cpp)
target_link_libraries(convtest blonda_core)
add_executable(songrender tools/songrender.cpp)
target_link_libraries(songrender blonda_core)
